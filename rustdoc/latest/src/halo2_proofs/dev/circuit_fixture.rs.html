26/// An [`Assignment`] that records the keygen-visible circuit layout without witnesses.
27#[derive(Debug)]
28pub struct CircuitFixtureRecorder<F: Field> {
29    in_region: bool,
30    regions: Vec<(String, Option<usize>)>,
31    copies: Vec<(Column<Any>, usize, Column<Any>, usize)>,
32    constants: Vec<(F, usize, usize)>,
33    fixed: Vec<(usize, usize, F)>,
34    fills: Vec<(usize, usize, F)>,
35    selectors: Vec<(usize, usize)>,
36}

38impl<F: Field> Default for CircuitFixtureRecorder<F> {
39    fn default() -> Self {

41            in_region: false,
42            regions: Vec::new(),
43            copies: Vec::new(),
44            constants: Vec::new(),
45            fixed: Vec::new(),
46            fills: Vec::new(),
47            selectors: Vec::new(),

52impl<F: Field> CircuitFixtureRecorder<F> {
53    fn touch(&mut self, row: usize) {
54        if self.in_region {
55            let minimum = &mut self.regions.last_mut().expect("entered region").1;
56            *minimum = Some(minimum.map_or(row, |current| current.min(row)));
57        }
58    }
59
60    fn selector_data(&self, meta: &ConstraintSystem<F>, n: usize) -> SelectorAssignmentData<F> {
61        let mut activations = vec![vec![false; n]; meta.lean_dump_num_selectors()];
62        for &(selector, row) in &self.selectors {
63            activations[selector][row] = true;
64        }
65        meta.lean_dump_selector_assignments(activations)
66    }
67}

69impl<F: Field> Assignment<F> for CircuitFixtureRecorder<F> {
70    fn enter_region<NR, N>(&mut self, name: N)
71    where
72        NR: Into<String>,
73        N: FnOnce() -> NR,
74    {
75        self.in_region = true;
76        self.regions.push((name().into(), None));
77    }
78
79    fn exit_region(&mut self) {
80        self.in_region = false;
81    }
82
83    fn enable_selector<A, AR>(&mut self, _: A, selector: &Selector, row: usize) -> Result<(), Error>
84    where
85        A: FnOnce() -> AR,
86        AR: Into<String>,
87    {
88        self.touch(row);
89        self.selectors.push((selector.0, row));
90        Ok(())
91    }
92
93    fn query_instance(&self, _: Column<Instance>, _: usize) -> Result<Value<F>, Error> {
94        Ok(Value::unknown())
95    }
96
97    fn assign_advice<V, VR, A, AR>(
98        &mut self,
99        _: A,
100        _: Column<Advice>,
101        row: usize,
102        _: V,
103    ) -> Result<(), Error>
104    where
105        V: FnOnce() -> Value<VR>,
106        VR: Into<Assigned<F>>,
107        A: FnOnce() -> AR,
108        AR: Into<String>,
109    {
110        self.touch(row);
111        Ok(())
112    }
113
114    fn assign_fixed<V, VR, A, AR>(
115        &mut self,
116        _: A,
117        column: Column<Fixed>,
118        row: usize,
119        value: V,
120    ) -> Result<(), Error>
121    where
122        V: FnOnce() -> Value<VR>,
123        VR: Into<Assigned<F>>,
124        A: FnOnce() -> AR,
125        AR: Into<String>,
126    {
127        self.touch(row);
128        let mut assigned = F::ZERO;
129        value().map(|value| assigned = Into::<Assigned<F>>::into(value).evaluate());
130        if !self.in_region {
131            self.constants.push((assigned, column.index(), row));
132        }
133        self.fixed.push((column.index(), row, assigned));
134        Ok(())
135    }
136
137    fn copy(
138        &mut self,
139        left_column: Column<Any>,
140        left_row: usize,
141        right_column: Column<Any>,
142        right_row: usize,
143    ) -> Result<(), Error> {
144        self.copies
145            .push((left_column, left_row, right_column, right_row));
146        Ok(())
147    }
148
149    fn fill_from_row(
150        &mut self,
151        column: Column<Fixed>,
152        row: usize,
153        value: Value<Assigned<F>>,
154    ) -> Result<(), Error> {
155        let mut assigned = F::ZERO;
156        value.map(|value| assigned = value.evaluate());
157        self.fills.push((column.index(), row, assigned));
158        Ok(())
159    }
160
161    fn push_namespace<NR, N>(&mut self, _: N)
162    where
163        NR: Into<String>,
164        N: FnOnce() -> NR,
165    {
166    }
167
168    fn pop_namespace(&mut self, _: Option<String>) {}
169}

172    /// Render the selector-compressed constraint system in Ironwood's compact JSON schema.
173    pub fn render_cs_json(&self, meta: &ConstraintSystem<F>, n: usize) -> String {
174        let mut activations = vec![vec![false; n]; meta.lean_dump_num_selectors()];
175        for &(selector, row) in &self.selectors {
176            activations[selector][row] = true;
177        }
178        meta.lean_dump_compressed(activations)
179            .lean_dump_cs_fixture_json(&field_decimal)
180    }

182    /// Render a complete keygen layout in Ironwood's compact JSON schema.
183    ///
184    /// `fill_column_order` preserves the order in which expanded fixed-column fills appear
185    /// in historical fixtures when that order differs from assignment order.
186    pub fn render_layout_json(
187        &self,
188        meta: &ConstraintSystem<F>,
189        k: u32,
190        fill_column_order: &[usize],
191    ) -> String {
192        let n = 1usize << k;
193        let permutation_columns = meta.lean_dump_permutation_columns();
194        let permutation_index = |column: &Column<Any>| {
195            permutation_columns
196                .iter()
197                .position(|candidate| candidate == column)
198                .expect("copied column must participate in the permutation")
199        };
200
201        let constants = self
202            .constants
203            .iter()
204            .map(|(value, column, row)| format!("[{},{},{}]", field_decimal(value), column, row))
205            .collect::<Vec<_>>()
206            .join(",");
207        let copies = self
208            .copies
209            .iter()
210            .map(|(left_column, left_row, right_column, right_row)| {
211                format!(
212                    "[{},{},{},{}]",
213                    permutation_index(left_column),
214                    left_row,
215                    permutation_index(right_column),
216                    right_row
217                )
218            })
219            .collect::<Vec<_>>()
220            .join(",");
221
222        let mut fixed = self.fixed.clone();
223        let usable_rows = n - (meta.lean_dump_blinding_factors() + 1);
224        let mut fills = self.fills.clone();
225        fills.sort_by_key(|(column, _, _)| {
226            fill_column_order
227                .iter()
228                .position(|candidate| candidate == column)
229                .expect("every filled column must have a serialization position")
230        });
231        for (column, first_row, value) in &fills {
232            for row in *first_row..usable_rows {
233                fixed.push((*column, row, *value));
234            }
235        }
236        let (fixed_before, packed, _) = self.selector_data(meta, n);
237        for (packed_index, values) in packed.iter().enumerate() {
238            for (row, value) in values.iter().enumerate() {
239                if *value != F::ZERO {
240                    fixed.push((fixed_before + packed_index, row, *value));
241                }
242            }
243        }
244        let fixed = fixed
245            .iter()
246            .map(|(column, row, value)| format!("[{},{},{}]", column, row, field_decimal(value)))
247            .collect::<Vec<_>>()
248            .join(",");
249
250        let columns = permutation_columns
251            .iter()
252            .map(|column| {
253                let kind = match column.column_type() {
254                    Any::Advice => "advice",
255                    Any::Fixed => "fixed",
256                    Any::Instance => "instance",
257                };
258                format!("[\"{}\",{}]", kind, column.index())
259            })
260            .collect::<Vec<_>>()
261            .join(",");
262        let regions = self
263            .regions
264            .iter()
265            .enumerate()
266            .map(|(index, (name, row))| {
267                // The Action circuit contains an empty region. Its historical JSON fixture
268                // represents the absent start row with zero, so preserve that sentinel.
269                format!("[{},{},{}]", index, json_string(name), row.unwrap_or(0))
270            })
271            .collect::<Vec<_>>()
272            .join(",");
273
274        let mut assembly = SigmaAssembly::new(permutation_columns.len(), n);
275        for (left_column, left_row, right_column, right_row) in &self.copies {
276            assembly.copy(
277                permutation_index(left_column),
278                *left_row,
279                permutation_index(right_column),
280                *right_row,
281            );
282        }
283        let mut sigma = Vec::new();
284        for column in 0..permutation_columns.len() {
285            for row in 0..n {
286                let mapped = assembly.mapping[column][row];
287                if mapped != (column, row) {
288                    sigma.push(format!("[{},{},{},{}]", column, row, mapped.0, mapped.1));
289                }
290            }
291        }
292
293        format!(
294            "{{\"constants\":[{}],\"copyList\":[{}],\"fixed\":[{}],\"k\":{},\"n\":{},\"permColumns\":[{}],\"regions\":[{}],\"sigma\":[{}]}}",
295            constants,
296            copies,
297            fixed,
298            k,
299            n,
300            columns,
301            regions,
302            sigma.join(","),
303        )
304    }

306    /// Render a selector-compression map in a caller-selected Lean environment.
307    pub fn render_selector_map_lean(
308        &self,
309        meta: &ConstraintSystem<F>,
310        n: usize,
311        header: &str,
312        environment: LeanEnvironment<'_>,
313        fixture_name: &str,
314        sort_assignments: bool,
315    ) -> String {
316        use std::fmt::Write as _;
317
318        let (fixed_before, packed, mut assignments) = self.selector_data(meta, n);
319        if sort_assignments {
320            assignments.sort_by_key(|(selector, _, _)| *selector);
321        }
322        let entries = assignments
323            .iter()
324            .map(|(selector, packed_index, root)| {
325                let len = assignments
326                    .iter()
327                    .filter(|(_, candidate, _)| candidate == packed_index)
328                    .count();
329                format!(
330                    "({}, ⟨{}, {}, {}⟩)",
331                    selector,
332                    fixed_before + packed_index,
333                    len,
334                    root
335                )
336            })
337            .collect::<Vec<_>>();
338
339        let mut output = format!(
340            "{}\nimport {}\n\nnamespace {}\n\nopen {}\n\ndef {} : SelCompressMap :=\n",
341            header,
342            environment.fixture_import,
343            environment.fixture_namespace,
344            environment.open_namespace,
345            fixture_name
346        );
347        writeln!(output, "  {{ newFixedCols := {},", packed.len()).unwrap();
348        if entries.is_empty() {
349            writeln!(output, "    entries := [] }}").unwrap();
350        } else {
351            writeln!(output, "    entries := [{}] }}", entries.join(",\n      ")).unwrap();
352        }
353        write!(output, "\nend {}\n", environment.fixture_namespace).unwrap();
354        output
355    }

380    /// Render a complete keygen layout in Ironwood's Lean fixture schema.
381    pub fn render_layout_lean(
382        &self,
383        meta: &ConstraintSystem<F>,
384        k: u32,
385        header: &str,
386        fixture_name: &str,
387        environment: LeanEnvironment<'_>,
388        long_data: bool,
389    ) -> String {
390        use std::fmt::Write as _;
391
392        let n = 1usize << k;
393        let permutation_columns = meta.lean_dump_permutation_columns();
394        let permutation_index = |column: &Column<Any>| {
395            permutation_columns
396                .iter()
397                .position(|candidate| candidate == column)
398                .expect("copied column must participate in the permutation")
399        };
400
401        let regions = self
402            .regions
403            .iter()
404            .enumerate()
405            .map(|(index, (name, row))| {
406                format!(
407                    "⟨{}, {:?}, {}⟩",
408                    index,
409                    name,
410                    row.expect("every recorded region must touch a row")
411                )
412            })
413            .collect::<Vec<_>>();
414        let columns = permutation_columns
415            .iter()
416            .map(|column| {
417                let kind = match column.column_type() {
418                    Any::Advice => "advice",
419                    Any::Fixed => "fixed",
420                    Any::Instance => "instance",
421                };
422                format!(".{} {}", kind, column.index())
423            })
424            .collect::<Vec<_>>();
425        let copies = self
426            .copies
427            .iter()
428            .map(|(left_column, left_row, right_column, right_row)| {
429                format!(
430                    "({}, {}, {}, {})",
431                    permutation_index(left_column),
432                    left_row,
433                    permutation_index(right_column),
434                    right_row
435                )
436            })
437            .collect::<Vec<_>>();
438
439        let mut assembly = SigmaAssembly::new(permutation_columns.len(), n);
440        for (left_column, left_row, right_column, right_row) in &self.copies {
441            assembly.copy(
442                permutation_index(left_column),
443                *left_row,
444                permutation_index(right_column),
445                *right_row,
446            );
447        }
448        let mut sigma = Vec::new();
449        for column in 0..permutation_columns.len() {
450            for row in 0..n {
451                let mapped = assembly.mapping[column][row];
452                if mapped != (column, row) {
453                    sigma.push(format!("({}, {}, {}, {})", column, row, mapped.0, mapped.1));
454                }
455            }
456        }
457
458        let constants = self
459            .constants
460            .iter()
461            .map(|(value, column, row)| format!("({}, {}, {})", field_decimal(value), column, row))
462            .collect::<Vec<_>>();
463        let mut fixed = self.fixed.clone();
464        let usable_rows = n - (meta.lean_dump_blinding_factors() + 1);
465        for (column, first_row, value) in &self.fills {
466            for row in *first_row..usable_rows {
467                fixed.push((*column, row, *value));
468            }
469        }
470        let (fixed_before, packed, _) = self.selector_data(meta, n);
471        for (packed_index, values) in packed.iter().enumerate() {
472            for (row, value) in values.iter().enumerate() {
473                if *value != F::ZERO {
474                    fixed.push((fixed_before + packed_index, row, *value));
475                }
476            }
477        }
478        fixed.sort_by_key(|(column, row, _)| (*column, *row));
479        let fixed = fixed
480            .iter()
481            .map(|(column, row, value)| format!("({}, {}, {})", column, row, field_decimal(value)))
482            .collect::<Vec<_>>();
483
484        let mut output = format!(
485            "{}\nnamespace {}\n\nopen {}\n\n",
486            header, environment.fixture_namespace, environment.open_namespace
487        );
488        if long_data {
489            output.push_str("-- `maxRecDepth` is raised only to elaborate the long flat data lists (the full lookup-table\n-- column contents can be thousands of rows). This is a data-literal elaboration depth, not a\n-- proof-search/heartbeat budget; the fixture is inert data.\nset_option maxRecDepth 100000 in\n");
490        }
491        writeln!(output, "def {fixture_name} : LayoutFixture :=").unwrap();
492        writeln!(output, "  {{ k := {k},").unwrap();
493        writeln!(output, "    n := {n},").unwrap();
494        writeln!(
495            output,
496            "    regions := {},",
497            lean_list(&regions).replace("\n      ", " ")
498        )
499        .unwrap();
500        writeln!(
501            output,
502            "    permColumns := {},",
503            lean_list(&columns).replace("\n      ", " ")
504        )
505        .unwrap();
506        writeln!(output, "    copyList := {},", lean_list(&copies)).unwrap();
507        writeln!(output, "    sigma := {},", lean_list(&sigma)).unwrap();
508        writeln!(
509            output,
510            "    constants := {},",
511            lean_list(&constants).replace("\n      ", " ")
512        )
513        .unwrap();
514        writeln!(output, "    fixed := {} }}", lean_list(&fixed)).unwrap();
515        write!(output, "\nend {}\n", environment.fixture_namespace).unwrap();
516        output
517    }
518}

528fn field_decimal<F: PrimeField>(value: &F) -> String {
529    const RADIX: u64 = 1_000_000_000;
530    let repr = value.to_repr();
531    let mut limbs = vec![0u32];
532    for &byte in repr.as_ref().iter().rev() {
533        let mut carry = byte as u64;
534        for limb in &mut limbs {
535            let next = (*limb as u64) * 256 + carry;
536            *limb = (next % RADIX) as u32;
537            carry = next / RADIX;
538        }
539        if carry != 0 {
540            limbs.push(carry as u32);
541        }
542    }
543    let mut output = format!("{}", limbs.pop().unwrap());
544    for limb in limbs.iter().rev() {
545        output.push_str(&format!("{limb:09}"));
546    }
547    output
548}

550fn json_string(value: &str) -> String {
551    use std::fmt::Write as _;
552
553    let mut output = String::from("\"");
554    for character in value.chars() {
555        match character {
556            '"' => output.push_str("\\\""),
557            '\\' => output.push_str("\\\\"),
558            '\n' => output.push_str("\\n"),
559            '\r' => output.push_str("\\r"),
560            '\t' => output.push_str("\\t"),
561            character if character <= '\u{1f}' => {
562                write!(output, "\\u{:04x}", character as u32).unwrap();
563            }
564            character => output.push(character),
565        }
566    }
567    output.push('"');
568    output
569}

571#[derive(Debug)]
572struct SigmaAssembly {
573    mapping: Vec<Vec<(usize, usize)>>,
574    auxiliary: Vec<Vec<(usize, usize)>>,
575    sizes: Vec<Vec<usize>>,
576}
577
578impl SigmaAssembly {
579    fn new(num_columns: usize, n: usize) -> Self {
580        let identity = (0..num_columns)
581            .map(|column| (0..n).map(|row| (column, row)).collect())
582            .collect::<Vec<Vec<_>>>();
583        Self {
584            mapping: identity.clone(),
585            auxiliary: identity,
586            sizes: vec![vec![1; n]; num_columns],
587        }
588    }
589
590    fn copy(&mut self, left_column: usize, left_row: usize, right_column: usize, right_row: usize) {
591        let mut left_cycle = self.auxiliary[left_column][left_row];
592        let mut right_cycle = self.auxiliary[right_column][right_row];
593        if left_cycle == right_cycle {
594            return;
595        }
596        if self.sizes[left_cycle.0][left_cycle.1] < self.sizes[right_cycle.0][right_cycle.1] {
597            std::mem::swap(&mut left_cycle, &mut right_cycle);
598        }
599        self.sizes[left_cycle.0][left_cycle.1] += self.sizes[right_cycle.0][right_cycle.1];
600        let mut position = right_cycle;
601        loop {
602            self.auxiliary[position.0][position.1] = left_cycle;
603            position = self.mapping[position.0][position.1];
604            if position == right_cycle {
605                break;
606            }
607        }
608        let left_mapping = self.mapping[left_column][left_row];
609        self.mapping[left_column][left_row] = self.mapping[right_column][right_row];
610        self.mapping[right_column][right_row] = left_mapping;
611    }
612}
