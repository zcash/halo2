1//! Lean fixture export for the verifier-fingerprint match (dev/test tooling).
2//!
3//! [`VerifyingKey::dump_vesta_lean_fixture`] emits a self-contained Lean file in the requested namespace,
4//! capturing one real proof: the verifying key (gates, query layouts, lookups, permutation
5//! chunking, domain), the proof string, the Fiat-Shamir challenges, and the assembled MSM, plus the
6//! `MsmMatch (assemble vk ps ch) capturedMsm := by native_decide` theorem.
7//!
8//! Scalar and base field elements are emitted as four little-endian `u64` limbs (`mkFp`/`mkFq`).
9//! Commitments and the complete verifier URS are emitted as validated concrete Vesta points, shared
10//! across the VK, proof string, transcript, and MSM.
11//!
12//! Trust boundary: the fixture attests that the assembled MSM matches — and, for accepting
13//! captures, evaluates to the identity — for *these* captured commitments and challenges. The
14//! emitted `vk` mirrors halo2's
15//! `VerifyingKey` field-for-field — circuit-fixed data only — so it stays faithful to the pinned Rust
16//! key; the instance commitment is *not* a VK field. Instead the fixture re-derives it from the
17//! public inputs (`commit_lagrange` of the zero-padded instance columns, exposed as
18//! `derivedInstanceCommitment`), feeds *that* to `assemble`, and checks it against the captured
19//! commitments (`instance_commitments_derived`), so instance commitments no longer enter as opaque
20//! points nor masquerade as VK data. It does not reproduce Halo2's Blake2b transcript or pinned-key
21//! serialization; those remain trusted from the Rust capture. Given the proof byte string the
22//! verifier consumed ([`VerifyingKey::dump_vesta_lean_fixture_honest_with_proof_bytes`]), it checks that
23//! re-serializing the recorded proof reads reproduces those bytes exactly and carries them in the
24//! fixture as `capturedProofHex`, so a consumer can check its own proof-string decoder against the
25//! bytes the deployed verifier read. Halo2's `MSM` also merges same-base
26//! terms and drops identity bases, where the Lean assembly deliberately does neither; such a capture
27//! is rejected at export (see [`VerifyingKey::dump_vesta_lean_fixture`]) rather than emitted.
28//!
29//! Two exporters share one emission path. [`VerifyingKey::dump_vesta_lean_fixture`] is the
30//! accepting exporter: it verifies the captured MSM is the group identity before emitting
31//! anything, so every honest fixture proves `capturedMsm.evalNat capturedURS = 0` alongside the
32//! match theorem. [`VerifyingKey::dump_vesta_lean_fixture_match_only`] is its sibling for
33//! non-accepting captures, of which random-input captures are the motivating case — the mode is
34//! selected by the captured MSM, so non-acceptance is what the export checks and randomness is the
35//! caller's business. The deployed verifier ran to completion, but the captured MSM is
36//! deliberately *not* the identity — verified before emitting anything — so the fixture emits the
37//! match theorem alongside `capturedMsm.evalNat capturedURS ≠ 0`, the mirror of the honest eval
38//! theorem. Either way the fixture proves its own mode rather than asserting it in a header
39//! comment. Rejecting runs of *honest* proofs remain Rust-only
40//! checks (they are non-identity / rejected by the deployed verifier), never exported to Lean:
41//! exporting a rejecting run as its own fixture was considered and dropped because it doubled the
42//! exporter surface for a cross-check that a trivially-accepting Lean `assemble` would already
43//! fail on the accepting fixtures.

59/// A field element as a Lean constructor call with four little-endian `u64` limbs.
60fn field<F: PrimeField>(constructor: &str, x: F) -> String {
61    let repr = x.to_repr();
62    let b = repr.as_ref();
63    debug_assert!(
64        b.len() <= 32,
65        "field repr is wider than the four u64 limbs emitted here; extra bytes would be silently dropped"
66    );
67    let limb = |i: usize| -> u64 {
68        let mut v: u64 = 0;
69        let mut j = 0;
70        while j < 8 {
71            let idx = i * 8 + j;
72            if idx < b.len() {
73                v |= (b[idx] as u64) << (8 * j);
74            }
75            j += 1;
76        }
77        v
78    };
79    format!(
80        "({} {} {} {} {})",
81        constructor,
82        limb(0),
83        limb(1),
84        limb(2),
85        limb(3)
86    )
87}

99/// Whether `segment` is a conservative ASCII subset of a Lean identifier: non-empty, starting with a
100/// letter or `_`, and otherwise containing only ASCII alphanumerics and `_`. A digit-leading segment
101/// (e.g. `123`) is rejected, so a dotted `lean_namespace` spliced verbatim into `namespace`/`end`
102/// cannot emit a token Lean would reject.
103fn is_lean_ident(segment: &str) -> bool {
104    let mut chars = segment.chars();
105    matches!(chars.next(), Some(c) if c.is_ascii_alphabetic() || c == '_')
106        && chars.all(|c| c.is_ascii_alphanumeric() || c == '_')
107}

109/// Coordinate key for a point's identity (its affine `(x, y)` repr; identity → sentinel).
110fn point_key(p: EqAffine) -> Vec<u8> {
111    let co: Option<Coordinates<EqAffine>> = p.coordinates().into();
112    match co {
113        Some(c) => {
114            let mut k = c.x().to_repr().as_ref().to_vec();
115            k.extend_from_slice(c.y().to_repr().as_ref());
116            k
117        }
118        None => vec![0u8],
119    }
120}

122/// Deduplicated concrete curve points used by the generated Lean fixture. The generated terms have
123/// type `VestaG`; the numeric indices are only source compression for repeated coordinate literals.
124struct PointTable {
125    ids: HashMap<Vec<u8>, usize>,
126    points: Vec<EqAffine>,
127}
128
129impl PointTable {
130    fn new() -> Self {
131        Self {
132            ids: HashMap::new(),
133            points: Vec::new(),
134        }
135    }
136
137    fn id(&mut self, point: EqAffine) -> usize {
138        let key = point_key(point);
139        if let Some(id) = self.ids.get(&key) {
140            *id
141        } else {
142            let id = self.points.len();
143            self.ids.insert(key, id);
144            self.points.push(point);
145            id
146        }
147    }
148
149    fn point_ref(&mut self, point: EqAffine) -> String {
150        format!("capturedPoint {}", self.id(point))
151    }
152
153    fn point_refs(&mut self, points: &[EqAffine]) -> Vec<String> {
154        points.iter().map(|point| self.point_ref(*point)).collect()
155    }
156
157    fn coordinate_literals(&self) -> Vec<String> {
158        self.points
159            .iter()
160            .map(|point| {
161                let coordinates: Option<Coordinates<EqAffine>> = point.coordinates().into();
162                match coordinates {
163                    Some(coordinates) => {
164                        format!("({}, {})", fq(*coordinates.x()), fq(*coordinates.y()))
165                    }
166                    None => "(0, 0)".to_string(),
167                }
168            })
169            .collect()
170    }
171}

192/// Lowercase hex, two digits per byte, no separators.
193fn hex_lower(bytes: &[u8]) -> String {
194    use std::fmt::Write as _;
195    let mut out = String::with_capacity(2 * bytes.len());
196    for b in bytes {
197        write!(out, "{b:02x}").expect("writing to a String cannot fail");
198    }
199    out
200}

202/// Re-serialize the proof reads the recorder observed, as the prover writes them: `write_point`
203/// (the compressed encoding) for each `ReadPoint` and `write_scalar` for each `ReadScalar`, in
204/// read order. Common inputs and squeezes consume no proof bytes.
205fn reserialize_proof_reads(events: &[TranscriptEvent<EqAffine>]) -> Vec<u8> {
206    let mut transcript = Blake2bWrite::<Vec<u8>, EqAffine, Challenge255<EqAffine>>::init(vec![]);
207    for event in events {
208        match event {
209            TranscriptEvent::ReadPoint(point) => transcript
210                .write_point(*point)
211                .expect("writing to an in-memory transcript cannot fail"),
212            TranscriptEvent::ReadScalar(scalar) => transcript
213                .write_scalar(*scalar)
214                .expect("writing to an in-memory transcript cannot fail"),
215            TranscriptEvent::CommonPoint(_)
216            | TranscriptEvent::CommonScalar(_)
217            | TranscriptEvent::Squeeze(_) => {}
218        }
219    }
220    transcript.finalize()
221}

223/// Serialize one gate / lookup `Expression` to a Lean `Expr Fp` literal (mirrors the verifier's
224/// `Expression::evaluate`; virtual selectors are removed before verification).
225fn expr_to_lean(e: &Expression<Fp>) -> String {
226    e.evaluate(
227        &|c| format!("(.constant {})", fp(c)),
228        &|_| panic!("virtual selectors are removed before verification"),
229        &|q| format!("(.fixed {})", q.index),
230        &|q| format!("(.advice {})", q.index),
231        &|q| format!("(.instance {})", q.index),
232        &|a: String| format!("(.negated {a})"),
233        &|a: String, b: String| format!("(.sum {a} {b})"),
234        &|a: String, b: String| format!("(.product {a} {b})"),
235        &|a: String, c| format!("(.scaled {} {})", a, fp(c)),
236    )
237}

239/// Render the typed transcript prefix and each subsequent squeeze checkpoint. `init_len` is the
240/// exact number of VK/instance events that precede verifier proof processing; it is intentionally
241/// independent of when the first proof element is read, because valid circuits may have no advice
242/// commitments before the first squeeze.
243fn render_transcript_capture(
244    points: &mut PointTable,
245    transcript_events: &[TranscriptEvent<EqAffine>],
246    init_len: usize,
247) -> (Vec<String>, Vec<String>) {
248    let mut captured_init = Vec::new();
249    let mut prefix = Vec::new();
250    let mut schedule_entries = Vec::new();
251
252    for (event_index, event) in transcript_events.iter().enumerate() {
253        let is_init = event_index < init_len;
254        match *event {
255            TranscriptEvent::CommonPoint(point) => {
256                let elt = transcript_point(points, point);
257                if is_init {
258                    captured_init.push(elt.clone());
259                }
260                prefix.push(elt);
261            }
262            TranscriptEvent::CommonScalar(scalar) => {
263                let elt = transcript_scalar(scalar);
264                if is_init {
265                    captured_init.push(elt.clone());
266                }
267                prefix.push(elt);
268            }
269            TranscriptEvent::ReadPoint(point) => {
270                assert!(
271                    !is_init,
272                    "proof read appeared inside the VK/instance prefix"
273                );
274                prefix.push(transcript_point(points, point));
275            }
276            TranscriptEvent::ReadScalar(scalar) => {
277                assert!(
278                    !is_init,
279                    "proof read appeared inside the VK/instance prefix"
280                );
281                prefix.push(transcript_scalar(scalar));
282            }
283            TranscriptEvent::Squeeze(challenge) => {
284                assert!(
285                    !is_init,
286                    "challenge squeeze appeared inside the VK/instance prefix"
287                );
288                schedule_entries.push(format!("([{}], {})", prefix.join(", "), fp(challenge)));
289                prefix.push(transcript_scalar(challenge));
290            }
291        }
292    }
293
294    (captured_init, schedule_entries)
295}

297/// Which theorem family the exporter emits, and which identity fail-fast it enforces.
298///
299/// Every dispatch on this is a `match` listing both variants, never an equality test with an
300/// implicit fallthrough: a new mode must fail to compile at each site until its fail-fast, its
301/// theorem set, and its header are all chosen. `#[non_exhaustive]` is inert while the enum is
302/// private and records the same intent for the day it is not.
303#[derive(Clone, Copy, Debug, PartialEq, Eq)]
304#[non_exhaustive]
305enum FixtureMode {
306    /// Accepting capture: the MSM must be the identity; the eval theorems are emitted.
307    Honest,
308    /// Non-accepting capture: the MSM must *not* be the identity, and the match theorem is
309    /// emitted alongside the mirrored `≠ 0` eval theorem.
310    MatchOnly,
311}

315    ///
316    /// `recorder` supplies the ordered transcript events, the proof elements (points and scalars in
317    /// read order), the instance commitments, and the squeezed challenges captured during the
318    /// verifier run; `captured_msm` is the assembled verifier fingerprint together with its exact
319    /// parameter set. Only *accepting* runs are exported: `captured_msm` is verified to be the group
320    /// identity before anything is emitted, so the fixture always proves `capturedMsm.evalNat = 0`.
321    ///
322    /// `instances` are the public inputs fed to the verifier for this run (`proof → column →

337        self.dump_vesta_lean_fixture_with_mode(
338            FixtureMode::Honest,
339            lean_namespace,
340            circuit_id,
341            k,
342            instances,
343            recorder,
344            captured_msm,
345            None,
346        )
347    }

443    // The public wrappers each carry the exporter's six or seven inputs; threading `mode` and the
444    // optional proof bytes through to the shared core adds two more past `&self`. Bundling them
445    // into a struct would only move the same arity to a constructor, since every one is required.
446    #[allow(clippy::too_many_arguments)]
447    fn dump_vesta_lean_fixture_with_mode<R: Read>(
448        &self,
449        mode: FixtureMode,
450        lean_namespace: &str,
451        circuit_id: &str,
452        k: u32,
453        instances: &[&[&[Fp]]],
454        recorder: &ChallengeRecorder<R, EqAffine, Challenge255<EqAffine>>,
455        captured_msm: &MSM<'_, EqAffine>,
456        proof_bytes: Option<&[u8]>,
457    ) -> String {
458        // `lean_namespace` is spliced verbatim into `namespace`/`end`, and `circuit_id` is emitted
459        // via `{:?}`, whose Rust string escapes (`\u{...}`, ...) are not Lean's. Validate both up
460        // front — in release builds too, since this runs once per export and a malformed name yields
461        // an uncompilable fixture. [`is_lean_ident`] accepts a conservative ASCII subset of Lean
462        // identifiers, so a digit-leading `lean_namespace` segment like `123` is rejected.
463        assert!(
464            lean_namespace.split('.').all(is_lean_ident),
465            "lean_namespace must be a dot-separated path of ASCII Lean identifiers \
466             (letter/underscore start): {lean_namespace:?}"
467        );
468        // `circuit_id` becomes a Lean *string literal*, not an identifier, so a leading digit is
469        // fine; it only has to stay a plain ASCII slug so `{:?}` cannot emit a `\u{...}` escape Lean
470        // would reject.
471        assert!(
472            !circuit_id.is_empty()
473                && circuit_id
474                    .chars()
475                    .all(|c| c.is_ascii_alphanumeric() || c == '_' || c == '-'),
476            "circuit_id must be a non-empty ASCII slug (alphanumeric, `_`, `-`): {circuit_id:?}"
477        );
478        // The supplied proof bytes must be exactly what the verifier's reads consumed: re-serializing
479        // the recorded reads as the prover writes them reproduces them byte for byte, so the typed
480        // proof emitted below is their canonical parse and no trailing data went unread.
481        if let Some(bytes) = proof_bytes {
482            let reserialized = reserialize_proof_reads(&recorder.events);
483            assert!(
484                reserialized == bytes,
485                "proof_bytes must be exactly the bytes the verifier's proof reads consume: \
486                 re-serializing the recorded reads gives {} bytes, the supplied string has {}",
487                reserialized.len(),
488                bytes.len()
489            );
490        }
491
492        // The public instances' outer dimension is the proof count, exactly as in `verify_proof`.
493        let num_proofs = instances.len();
494
495        let common_points = recorder.common_points.as_slice();
496        let read_points = recorder.points.as_slice();
497        let read_scalars = recorder.scalars.as_slice();
498        let challenges = recorder.challenges.as_slice();
499        let transcript_events = recorder.events.as_slice();
500
501        let params = captured_msm.params;
502        // The honest mode's `capturedMsm_eval_eq_zero` presupposes an accepting capture, and the
503        // match-only mode's `capturedMsm_evalNat_ne_zero` presupposes the opposite; fail fast on a
504        // mode/capture mismatch rather than export a fixture whose theorem cannot hold, or one that
505        // mislabels an accepting run as match-only. This is the only thing that selects the mode:
506        // whether the caller's input was a random proof string is neither checked nor recorded.
507        // Rejecting runs of honest proofs are checked in Rust, never exported to Lean.
508        match mode {
509            FixtureMode::Honest => assert!(
510                captured_msm.clone().eval(),
511                "captured MSM must evaluate to the identity; the emitted fixture proves it"
512            ),
513            FixtureMode::MatchOnly => assert!(
514                !captured_msm.clone().eval(),
515                "match-only fixtures capture non-accepting runs, but this captured MSM evaluates \
516                 to the identity; export accepting captures with `dump_vesta_lean_fixture`"
517            ),
518        }
519        let (msm_g, msm_w, msm_u, msm_other) = captured_msm.fingerprint_terms();
520        let msm_g = msm_g.expect("captured MSM must contain verifier-generator coefficients");
521        let msm_w = msm_w.expect("captured MSM must contain a blinding-generator coefficient");
522        let msm_u = msm_u.expect("captured MSM must contain an IPA-generator coefficient");
523        let cs = &self.cs;
524        let n_advice = cs.num_advice_columns;
525        let n_inst_cols = cs.num_instance_columns;
526        let n_lookups = cs.lookups.len();
527        let chunk_len = self.cs_degree - 2;
528        let perm_columns = cs.permutation.get_columns();
529        // A zero chunk length is only consistent with there being no permutation columns; otherwise
530        // `n_perm_sets` below and the chunk-export loop (which falls back to `chunk_len.max(1)`)
531        // would disagree on how many sets exist.
532        assert!(
533            chunk_len > 0 || perm_columns.is_empty(),
534            "permutation columns present but chunk length is zero"
535        );
536        let n_perm_sets = if chunk_len == 0 {
537            0
538        } else {
539            perm_columns.chunks(chunk_len).count()
540        };
541        let n_perm_cols = self.permutation.commitments().len();
542        let n_quotient = self.domain.get_quotient_poly_degree();
543        let n_inst_q = cs.instance_queries.len();
544        let n_adv_q = cs.advice_queries.len();
545        let n_fixed_q = cs.fixed_queries.len();
546        let blinding = cs.blinding_factors();
547        let n: u64 = 1u64 << k;
548        assert_eq!(params.k, k);
549        assert_eq!(params.g.len(), n as usize);
