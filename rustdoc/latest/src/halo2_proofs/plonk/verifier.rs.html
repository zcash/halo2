21// Batch normalization costs more than individual normalization below this.
22const MIN_BATCH_NORMALIZE: usize = 4;
23
24/// Trait representing a strategy for verifying Halo 2 proofs.
25pub trait VerificationStrategy<'params, C: CurveAffine> {
26    /// The output type of this verification strategy after processing a proof.
27    type Output;
28
29    /// Obtains an MSM from the verifier strategy and yields back the strategy's
30    /// output.
31    fn process<E: EncodedChallenge<C>>(
32        self,
33        f: impl FnOnce(MSM<'params, C>) -> Result<Guard<'params, C, E>, Error>,
34    ) -> Result<Self::Output, Error>;
35}
36
37/// A verifier that checks a single proof at a time.
38#[derive(Debug)]
39pub struct SingleVerifier<'params, C: CurveAffine> {
40    msm: MSM<'params, C>,
41}
42
43impl<'params, C: CurveAffine> SingleVerifier<'params, C> {
44    /// Constructs a new single proof verifier.
45    pub fn new(params: &'params Params<C>) -> Self {
46        SingleVerifier {
47            msm: MSM::new(params),
48        }
49    }
50}
51
52impl<'params, C: CurveAffine> VerificationStrategy<'params, C> for SingleVerifier<'params, C> {
53    type Output = ();
54
55    fn process<E: EncodedChallenge<C>>(
56        self,
57        f: impl FnOnce(MSM<'params, C>) -> Result<Guard<'params, C, E>, Error>,
58    ) -> Result<Self::Output, Error> {
59        let guard = f(self.msm)?;
60        let msm = guard.use_challenges();
61        if msm.eval() {
62            Ok(())
63        } else {
64            Err(Error::ConstraintSystemFailure)
65        }
66    }
67}

69/// Returns a boolean indicating whether or not the proof is valid
70pub fn verify_proof<
71    'params,
72    C: CurveAffine,
73    E: EncodedChallenge<C>,
74    T: TranscriptRead<C, E>,
75    V: VerificationStrategy<'params, C>,
76>(
77    params: &'params Params<C>,
78    vk: &VerifyingKey<C>,
79    strategy: V,
80    instances: &[&[&[C::Scalar]]],
81    transcript: &mut T,
82) -> Result<V::Output, Error> {
83    // Check that instances matches the expected number of instance columns
84    for instances in instances.iter() {
85        if instances.len() != vk.cs.num_instance_columns {
86            return Err(Error::InvalidInstances);
87        }
88    }
89
90    let max_instance_len = params.n as usize - (vk.cs.blinding_factors() + 1);
91    if instances
92        .iter()
93        .flat_map(|instance| instance.iter())
94        .any(|instance| instance.len() > max_instance_len)
95    {
96        return Err(Error::InstanceTooLarge);
97    }
98
99    let batch_normalize = instances
100        .iter()
101        .flat_map(|instance| instance.iter())
102        .take(MIN_BATCH_NORMALIZE)
103        .count()
104        == MIN_BATCH_NORMALIZE;
105    let instance_commitments = if !batch_normalize {
106        instances
107            .iter()
108            .map(|instance| {
109                instance
110                    .iter()
111                    .map(|instance| {
112                        params
113                            .commit_lagrange_instance(instance)
114                            .to_affine_vartime()
115                    })
116                    .collect::<Vec<_>>()
117            })
118            .collect::<Vec<_>>()
119    } else {
120        let instance_commitments_projective = instances
121            .iter()
122            .flat_map(|instance| instance.iter())
123            .map(|instance| params.commit_lagrange_instance(instance))
124            .collect::<Vec<_>>();
125        let mut normalized_commitments = vec![C::identity(); instance_commitments_projective.len()];
126        C::Curve::batch_normalize_vartime(
127            &instance_commitments_projective,
128            &mut normalized_commitments,
129        );
130        let mut normalized_commitments = normalized_commitments.into_iter();
131        let instance_commitments = instances
132            .iter()
133            .map(|instance| {
134                normalized_commitments
135                    .by_ref()
136                    .take(instance.len())
137                    .collect::<Vec<_>>()
138            })
139            .collect::<Vec<_>>();
140        assert!(normalized_commitments.next().is_none());
141        instance_commitments
142    };
143
144    let num_proofs = instance_commitments.len();
145
146    // Hash verification key into transcript
147    vk.hash_into(transcript)?;
148
149    for instance_commitments in instance_commitments.iter() {
150        // Hash the instance (external) commitments into the transcript
151        for commitment in instance_commitments {
152            transcript.common_point(*commitment)?
153        }
154    }
155
156    let advice_commitments = (0..num_proofs)
157        .map(|_| -> Result<Vec<_>, _> {
158            // Hash the prover's advice commitments into the transcript
159            read_n_points(transcript, vk.cs.num_advice_columns)
160        })
161        .collect::<Result<Vec<_>, _>>()?;
162
163    // Sample theta challenge for keeping lookup columns linearly independent
164    let theta: ChallengeTheta<_> = transcript.squeeze_challenge_scalar();
165
166    let lookups_permuted = (0..num_proofs)
167        .map(|_| -> Result<Vec<_>, _> {
168            // Hash each lookup permuted commitment
169            vk.cs
170                .lookups
171                .iter()
172                .map(|argument| argument.read_permuted_commitments(transcript))
173                .collect::<Result<Vec<_>, _>>()
174        })
175        .collect::<Result<Vec<_>, _>>()?;
176
177    // Sample beta challenge
178    let beta: ChallengeBeta<_> = transcript.squeeze_challenge_scalar();
179
180    // Sample gamma challenge
181    let gamma: ChallengeGamma<_> = transcript.squeeze_challenge_scalar();
182
183    let permutations_committed = (0..num_proofs)
184        .map(|_| {
185            // Hash each permutation product commitment
186            vk.cs.permutation.read_product_commitments(vk, transcript)
187        })
188        .collect::<Result<Vec<_>, _>>()?;
189
190    let lookups_committed = lookups_permuted
191        .into_iter()
192        .map(|lookups| {
193            // Hash each lookup product commitment
194            lookups
195                .into_iter()
196                .map(|lookup| lookup.read_product_commitment(transcript))
197                .collect::<Result<Vec<_>, _>>()
198        })
199        .collect::<Result<Vec<_>, _>>()?;
200
201    let vanishing = vanishing::Argument::read_commitments_before_y(transcript)?;
202
203    // Sample y challenge, which keeps the gates linearly independent.
204    let y: ChallengeY<_> = transcript.squeeze_challenge_scalar();
205
206    let vanishing = vanishing.read_commitments_after_y(vk, transcript)?;
207
208    // Sample x challenge, which is used to ensure the circuit is
209    // satisfied with high probability.
210    let x: ChallengeX<_> = transcript.squeeze_challenge_scalar();
211    let instance_evals = (0..num_proofs)
212        .map(|_| -> Result<Vec<_>, _> { read_n_scalars(transcript, vk.cs.instance_queries.len()) })
213        .collect::<Result<Vec<_>, _>>()?;
214
215    let advice_evals = (0..num_proofs)
216        .map(|_| -> Result<Vec<_>, _> { read_n_scalars(transcript, vk.cs.advice_queries.len()) })
217        .collect::<Result<Vec<_>, _>>()?;
218
219    let fixed_evals = read_n_scalars(transcript, vk.cs.fixed_queries.len())?;
220
221    let vanishing = vanishing.evaluate_after_x(transcript)?;
222
223    let permutations_common = vk.permutation.evaluate(transcript)?;
224
225    let permutations_evaluated = permutations_committed
226        .into_iter()
227        .map(|permutation| permutation.evaluate(transcript))
228        .collect::<Result<Vec<_>, _>>()?;
229
230    let lookups_evaluated = lookups_committed
231        .into_iter()
232        .map(|lookups| -> Result<Vec<_>, _> {
233            lookups
234                .into_iter()
235                .map(|lookup| lookup.evaluate(transcript))
236                .collect::<Result<Vec<_>, _>>()
237        })
238        .collect::<Result<Vec<_>, _>>()?;
239
240    // This check ensures the circuit is satisfied so long as the polynomial
241    // commitments open to the correct values.
242    let vanishing = {
243        // x^n
244        let xn = x.pow([params.n, 0, 0, 0]);
245
246        let blinding_factors = vk.cs.blinding_factors();
247        let l_evals = vk
248            .domain
249            .l_i_range(*x, xn, (-((blinding_factors + 1) as i32))..=0);
250        assert_eq!(l_evals.len(), 2 + blinding_factors);
251        let l_last = l_evals[0];
252        let l_blind: C::Scalar = l_evals[1..(1 + blinding_factors)]
253            .iter()
254            .fold(C::Scalar::ZERO, |acc, eval| acc + eval);
255        let l_0 = l_evals[1 + blinding_factors];
256
257        // Compute the expected value of h(x)
258        let expressions = advice_evals
259            .iter()
260            .zip(instance_evals.iter())
261            .zip(permutations_evaluated.iter())
262            .zip(lookups_evaluated.iter())
263            .flat_map(|(((advice_evals, instance_evals), permutation), lookups)| {
264                let fixed_evals = &fixed_evals;
265                std::iter::empty()
266                    // Evaluate the circuit using the custom gates provided
267                    .chain(vk.cs.gates.iter().flat_map(move |gate| {
268                        gate.polynomials().iter().map(move |poly| {
269                            poly.evaluate(
270                                &|scalar| scalar,
271                                &|_| panic!("virtual selectors are removed during optimization"),
272                                &|query| fixed_evals[query.index],
273                                &|query| advice_evals[query.index],
274                                &|query| instance_evals[query.index],
275                                &|a| -a,
276                                &|a, b| a + &b,
277                                &|a, b| a * &b,
278                                &|a, scalar| a * &scalar,
279                            )
280                        })
281                    }))
282                    .chain(permutation.expressions(
283                        vk,
284                        &vk.cs.permutation,
285                        &permutations_common,
286                        advice_evals,
287                        fixed_evals,
288                        instance_evals,
289                        l_0,
290                        l_last,
291                        l_blind,
292                        beta,
293                        gamma,
294                        x,
295                    ))
296                    .chain(lookups.iter().zip(vk.cs.lookups.iter()).flat_map(
297                        move |(p, argument)| {
298                            p.expressions(
299                                l_0,
300                                l_last,
301                                l_blind,
302                                argument,
303                                theta,
304                                beta,
305                                gamma,
306                                advice_evals,
307                                fixed_evals,
308                                instance_evals,
309                            )
310                        },
311                    ))
312            });
313
314        vanishing.verify(params, expressions, y, xn)
315    };
316
317    let queries = instance_commitments
318        .iter()
319        .zip(instance_evals.iter())
320        .zip(advice_commitments.iter())
321        .zip(advice_evals.iter())
322        .zip(permutations_evaluated.iter())
323        .zip(lookups_evaluated.iter())
324        .flat_map(
325            |(
326                (
327                    (((instance_commitments, instance_evals), advice_commitments), advice_evals),
328                    permutation,
329                ),
330                lookups,
331            )| {
332                iter::empty()
333                    .chain(vk.cs.instance_queries.iter().enumerate().map(
334                        move |(query_index, &(column, at))| {
335                            VerifierQuery::new_commitment(
336                                &instance_commitments[column.index()],
337                                vk.domain.rotate_omega(*x, at),
338                                instance_evals[query_index],
339                            )
340                        },
341                    ))
342                    .chain(vk.cs.advice_queries.iter().enumerate().map(
343                        move |(query_index, &(column, at))| {
344                            VerifierQuery::new_commitment(
345                                &advice_commitments[column.index()],
346                                vk.domain.rotate_omega(*x, at),
347                                advice_evals[query_index],
348                            )
349                        },
350                    ))
351                    .chain(permutation.queries(vk, x))
352                    .chain(lookups.iter().flat_map(move |p| p.queries(vk, x)))
353            },
354        )
355        .chain(
356            vk.cs
357                .fixed_queries
358                .iter()
359                .enumerate()
360                .map(|(query_index, &(column, at))| {
361                    VerifierQuery::new_commitment(
362                        &vk.fixed_commitments[column.index()],
363                        vk.domain.rotate_omega(*x, at),
364                        fixed_evals[query_index],
365                    )
366                }),
367        )
368        .chain(permutations_common.queries(&vk.permutation, x))
369        .chain(vanishing.queries(x));
370
371    // We are now convinced the circuit is satisfied so long as the
372    // polynomial commitments open to the correct values.
373    strategy.process(|msm| {
374        multiopen::verify_proof(params, transcript, queries, msm).map_err(|_| Error::Opening)
375    })
376}

443    fn instance_commitment_matches_zero_padded_commitment() {
444        const K: u32 = 5;
445
446        let params = Params::<EqAffine>::new(K);
447        let domain = EvaluationDomain::new(1, K);
448
449        for len in [0, 1, 9, 1 << K] {
450            let instance = (0..len).map(|i| Fp::from(i as u64)).collect::<Vec<_>>();
451            let mut padded = domain.empty_lagrange();
452            for (coefficient, value) in padded.iter_mut().zip(instance.iter()) {
453                *coefficient = *value;
454            }
455
456            assert_eq!(
457                params.commit_lagrange_instance(&instance),
458                params.commit_lagrange(&padded, Blind::default()),
459            );
460        }
