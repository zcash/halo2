25/// Number of bits for which complete addition needs to be used in variable-base
26/// scalar multiplication
27const NUM_COMPLETE_BITS: usize = 3;
28
29// Bits used in incomplete addition. k_{254} to k_{4} inclusive
30const INCOMPLETE_LEN: usize = pallas::Scalar::NUM_BITS as usize - 1 - NUM_COMPLETE_BITS;
31
32// Bits k_{254} to k_{4} inclusive are used in incomplete addition.
33// The `hi` half is k_{254} to k_{130} inclusive (length 125 bits).
34// (It is a coincidence that k_{130} matches the boundary of the
35// overflow check described in [the book](https://zcash.github.io/halo2/design/gadgets/ecc/var-base-scalar-mul.html#overflow-check).)
36const INCOMPLETE_HI_RANGE: Range<usize> = 0..INCOMPLETE_HI_LEN;
37const INCOMPLETE_HI_LEN: usize = INCOMPLETE_LEN / 2;
38
39// Bits k_{254} to k_{4} inclusive are used in incomplete addition.
40// The `lo` half is k_{129} to k_{4} inclusive (length 126 bits).
41const INCOMPLETE_LO_RANGE: Range<usize> = INCOMPLETE_HI_LEN..INCOMPLETE_LEN;
42const INCOMPLETE_LO_LEN: usize = INCOMPLETE_LEN - INCOMPLETE_HI_LEN;
43
44// Bits k_{3} to k_{1} inclusive are used in complete addition.
45// Bit k_{0} is handled separately.
46const COMPLETE_RANGE: Range<usize> = INCOMPLETE_LEN..(INCOMPLETE_LEN + NUM_COMPLETE_BITS);

65    pub(super) fn configure(
66        meta: &mut ConstraintSystem<pallas::Base>,
67        add_config: add::Config,
68        lookup_config: Lookup,
69        advices: [Column<Advice>; 10],
70    ) -> Self {
71        let hi_config = incomplete::Config::configure(
72            meta, advices[9], advices[3], advices[0], advices[1], advices[4], advices[5],
73        );
74        let lo_config = incomplete::Config::configure(
75            meta, advices[6], advices[7], advices[0], advices[1], advices[8], advices[2],
76        );
77        let complete_config = complete::Config::configure(meta, advices[9], add_config);
78        let overflow_config =
79            overflow::Config::configure(meta, lookup_config, advices[6..9].try_into().unwrap());
80
81        let config = Self {
82            q_mul_lsb: meta.selector(),
83            add_config,
84            hi_config,
85            lo_config,
86            complete_config,
87            overflow_config,
88        };
89
90        config.create_gate(meta);
91
92        assert_eq!(
93            config.hi_config.double_and_add.x_p, config.lo_config.double_and_add.x_p,
94            "x_p is shared across hi and lo halves."
95        );
96        assert_eq!(
97            config.hi_config.y_p, config.lo_config.y_p,
98            "y_p is shared across hi and lo halves."
99        );
100
101        // For both hi_config and lo_config:
102        // z and lambda1 are assigned on the same row as the add_config output.
103        // Therefore, z and lambda1 must not overlap with add_config.x_qr, add_config.y_qr.
104        let add_config_outputs = config.add_config.output_columns();
105        {
106            assert!(
107                !add_config_outputs.contains(&config.hi_config.z),
108                "incomplete config z cannot overlap with complete addition columns."
109            );
110            assert!(
111                !add_config_outputs.contains(&config.hi_config.double_and_add.lambda_1),
112                "incomplete config lambda1 cannot overlap with complete addition columns."
113            );
114        }
115        {
116            assert!(
117                !add_config_outputs.contains(&config.lo_config.z),
118                "incomplete config z cannot overlap with complete addition columns."
119            );
120            assert!(
121                !add_config_outputs.contains(&config.lo_config.double_and_add.lambda_1),
122                "incomplete config lambda1 cannot overlap with complete addition columns."
123            );
124        }
125
126        config
127    }

129    fn create_gate(&self, meta: &mut ConstraintSystem<pallas::Base>) {
130        // If `lsb` is 0, (x, y) = (x_p, -y_p). If `lsb` is 1, (x, y) = (0,0).
131        // https://p.z.cash/halo2-0.1:ecc-var-mul-lsb-gate?partial
132        meta.create_gate("LSB check", |meta| {
133            let q_mul_lsb = meta.query_selector(self.q_mul_lsb);
134
135            let z_1 = meta.query_advice(self.complete_config.z_complete, Rotation::cur());
136            let z_0 = meta.query_advice(self.complete_config.z_complete, Rotation::next());
137            let x_p = meta.query_advice(self.add_config.x_p, Rotation::cur());
138            let y_p = meta.query_advice(self.add_config.y_p, Rotation::cur());
139            let base_x = meta.query_advice(self.add_config.x_p, Rotation::next());
140            let base_y = meta.query_advice(self.add_config.y_p, Rotation::next());
141
142            //    z_0 = 2 * z_1 + k_0
143            // => k_0 = z_0 - 2 * z_1
144            let lsb = z_0 - z_1 * pallas::Base::from(2);
145
146            let bool_check = bool_check(lsb.clone());
147
148            // `lsb` = 0 => (x_p, y_p) = (x, -y)
149            // `lsb` = 1 => (x_p, y_p) = (0,0)
150            let lsb_x = ternary(lsb.clone(), x_p.clone(), x_p - base_x);
151            let lsb_y = ternary(lsb, y_p.clone(), y_p + base_y);
152
153            Constraints::with_selector(
154                q_mul_lsb,
155                [
156                    ("bool_check", bool_check),
157                    ("lsb_x", lsb_x),
158                    ("lsb_y", lsb_y),
159                ],
160            )
161        });
162    }

164    pub(super) fn assign(
165        &self,
166        mut layouter: impl Layouter<pallas::Base>,
167        alpha: AssignedCell<pallas::Base, pallas::Base>,
168        base: &NonIdentityEccPoint,
169        circuit_version: CircuitVersion,
170    ) -> Result<(EccPoint, ScalarVar), Error> {
171        let (result, zs): (EccPoint, Vec<Z<pallas::Base>>) = layouter.assign_region(
172            || "variable-base scalar mul",
173            |mut region| {
174                let offset = 0;
175
176                // Case `base` into an `EccPoint` for later use.
177                let base_point: EccPoint = base.clone().into();
178
179                // Decompose `k = alpha + t_q` bitwise (big-endian bit order).
180                let bits = decompose_for_scalar_mul(alpha.value());
181
182                // Define ranges for each part of the algorithm.
183                let bits_incomplete_hi = &bits[INCOMPLETE_HI_RANGE];
184                let bits_incomplete_lo = &bits[INCOMPLETE_LO_RANGE];
185                let lsb = bits[pallas::Scalar::NUM_BITS as usize - 1];
186
187                // Initialize the accumulator `acc = [2]base` using complete addition.
188                let acc =
189                    self.add_config
190                        .assign_region(&base_point, &base_point, offset, &mut region)?;
191
192                // Increase the offset by 1 after complete addition.
193                let offset = offset + 1;
194
195                // Initialize the running sum for scalar decomposition to zero.
196                //
197                // `incomplete::Config::double_and_add` will copy this cell directly into
198                // itself. This is fine because we are just assigning the same value to
199                // the same cell twice, and then applying an equality constraint between
200                // the cell and itself (which the permutation argument treats as a no-op).
201                let z_init = Z(region.assign_advice_from_constant(
202                    || "z_init = 0",
203                    self.hi_config.z,
204                    offset,
205                    pallas::Base::zero(),
206                )?);
207
208                // Double-and-add (incomplete addition) for the `hi` half of the scalar decomposition
209                let (x_a, y_a, zs_incomplete_hi) = self.hi_config.double_and_add(
210                    &mut region,
211                    offset,
212                    base,
213                    bits_incomplete_hi,
214                    (X(acc.x), Y(acc.y), z_init.clone()),
215                    circuit_version,
216                )?;
217
218                // Double-and-add (incomplete addition) for the `lo` half of the scalar decomposition
219                let z = zs_incomplete_hi.last().expect("should not be empty");
220                let (x_a, y_a, zs_incomplete_lo) = self.lo_config.double_and_add(
221                    &mut region,
222                    offset,
223                    base,
224                    bits_incomplete_lo,
225                    (x_a, y_a, z.clone()),
226                    circuit_version,
227                )?;
228
229                // Move from incomplete addition to complete addition.
230                // Inside incomplete::double_and_add, the offset was increased once after initialization
231                // of the running sum.
232                // Then, the final assignment of double-and-add was made on row + offset + 1.
233                // Outside of incomplete addition, we must account for these offset increases by adding
234                // 2 to the incomplete addition length.
235                assert!(INCOMPLETE_LO_RANGE.len() >= INCOMPLETE_HI_RANGE.len());
236                let offset = offset + INCOMPLETE_LO_RANGE.len() + 2;
237
238                // Complete addition
239                let (acc, zs_complete) = {
240                    let z = zs_incomplete_lo.last().expect("should not be empty");
241                    // Bits used in complete addition. k_{3} to k_{1} inclusive
242                    // The LSB k_{0} is handled separately.
243                    let bits_complete = &bits[COMPLETE_RANGE];
244                    self.complete_config.assign_region(
245                        &mut region,
246                        offset,
247                        bits_complete,
248                        &base_point,
249                        x_a,
250                        y_a,
251                        z.clone(),
252                    )?
253                };
254
255                // Each iteration of the complete addition uses two rows.
256                let offset = offset + COMPLETE_RANGE.len() * 2;
257
258                // Process the least significant bit
259                let z_1 = zs_complete.last().unwrap().clone();
260                let (result, z_0) = self.process_lsb(&mut region, offset, base, acc, z_1, lsb)?;
261
262                #[cfg(test)]
263                // Check that the correct multiple is obtained.
264                {
265                    use group::Curve;
266
267                    let base = base.point();
268                    let alpha = alpha
269                        .value()
270                        .map(|alpha| pallas::Scalar::from_repr(alpha.to_repr()).unwrap());
271                    let real_mul = base.zip(alpha).map(|(base, alpha)| base * alpha);
272                    let result = result.point();
273
274                    real_mul
275                        .zip(result)
276                        .assert_if_known(|(real_mul, result)| &real_mul.to_affine() == result);
277                }
278
279                let zs = {
280                    let mut zs = std::iter::empty()
281                        .chain(Some(z_init))
282                        .chain(zs_incomplete_hi)
283                        .chain(zs_incomplete_lo)
284                        .chain(zs_complete)
285                        .chain(Some(z_0))
286                        .collect::<Vec<_>>();
287                    assert_eq!(zs.len(), pallas::Scalar::NUM_BITS as usize + 1);
288
289                    // This reverses zs to give us [z_0, z_1, ..., z_{254}, z_{255}].
290                    zs.reverse();
291                    zs
292                };
293
294                Ok((result, zs))
295            },
296        )?;
297
298        self.overflow_config.overflow_check(
299            layouter.namespace(|| "overflow check"),
300            alpha.clone(),
301            &zs,
302        )?;
303
304        Ok((result, ScalarVar::BaseFieldElem(alpha)))
305    }
306
307    /// Processes the final scalar bit `k_0`.
308    ///
309    /// Assumptions for this sub-region:
310    /// - `acc_x` and `acc_y` are assigned in row `offset` by the previous complete
311    ///   addition. They will be copied into themselves.
312    /// - `z_1 is assigned in row `offset` by the mul::complete region assignment. We only
313    ///   use its value here.
314    ///
315    /// `x_p` and `y_p` are assigned here, and then copied into themselves by the complete
316    /// addition subregion.
317    ///
318    /// ```text
319    /// | x_p  | y_p  | acc_x | acc_y | complete addition  | z_1 | q_mul_lsb = 1
320    /// |base_x|base_y| res_x | res_y |   |   |    |   |   | z_0 |
321    /// ```
322    ///
323    /// [Specification](https://p.z.cash/halo2-0.1:ecc-var-mul-lsb-gate?partial).
324    fn process_lsb(
325        &self,
326        region: &mut Region<'_, pallas::Base>,
327        offset: usize,
328        base: &NonIdentityEccPoint,
329        acc: EccPoint,
330        z_1: Z<pallas::Base>,
331        lsb: Value<bool>,
332    ) -> Result<(EccPoint, Z<pallas::Base>), Error> {
333        // Enforce switching logic on LSB using a custom gate
334        self.q_mul_lsb.enable(region, offset)?;
335
336        // z_1 has been assigned at (z_complete, offset).
337        // Assign z_0 = 2⋅z_1 + k_0
338        let z_0 = {
339            let z_0_val = z_1.value().zip(lsb).map(|(z_1, lsb)| {
340                let lsb = pallas::Base::from(lsb as u64);
341                z_1 * pallas::Base::from(2) + lsb
342            });
343            let z_0_cell = region.assign_advice(
344                || "z_0",
345                self.complete_config.z_complete,
346                offset + 1,
347                || z_0_val,
348            )?;
349
350            Z(z_0_cell)
351        };
352
353        // Copy in `base_x`, `base_y` to use in the LSB gate
354        base.x()
355            .copy_advice(|| "copy base_x", region, self.add_config.x_p, offset + 1)?;
356        base.y()
357            .copy_advice(|| "copy base_y", region, self.add_config.y_p, offset + 1)?;
358
359        // If `lsb` is 0, return `Acc + (-P)`. If `lsb` is 1, simply return `Acc + 0`.
360        let x = lsb.and_then(|lsb| {
361            if !lsb {
362                base.x.value().cloned()
363            } else {
364                Value::known(Assigned::Zero)
365            }
366        });
367
368        let y = lsb.and_then(|lsb| {
369            if !lsb {
370                -base.y.value()
371            } else {
372                Value::known(Assigned::Zero)
373            }
374        });
375
376        let x_cell = region.assign_advice(|| "x", self.add_config.x_p, offset, || x)?;
377        let y_cell = region.assign_advice(|| "y", self.add_config.y_p, offset, || y)?;
378
379        let p = EccPoint::from_coordinates_unchecked(x_cell, y_cell);
380
381        // Return the result of the final complete addition as `[scalar]B`
382        let result = self.add_config.assign_region(&p, &acc, offset, region)?;
383
384        Ok((result, z_0))
385    }
386}

421// https://p.z.cash/halo2-0.1:ecc-var-mul-witness-scalar?partial
422#[allow(clippy::assign_op_pattern)]
423#[allow(clippy::ptr_offset_with_cast)]
424#[allow(clippy::manual_div_ceil)]
425fn decompose_for_scalar_mul(scalar: Value<&pallas::Base>) -> Vec<Value<bool>> {
426    construct_uint! {
427        struct U256(4);
428    }
429
430    let bitstring = scalar.map(|scalar| {
431        // We use `k = scalar + t_q` in the double-and-add algorithm, where
432        // the scalar field `F_q = 2^254 + t_q`.
433        // Note that the addition `scalar + t_q` is not reduced.
434        //
435        let scalar = U256::from_little_endian(&scalar.to_repr());
436        let t_q = U256::from_little_endian(&T_Q.to_le_bytes());
437        let k = scalar + t_q;
438
439        // Little-endian bit representation of `k`.
440        let bitstring = {
441            let mut le_bytes = [0u8; 32];
442            k.to_little_endian(&mut le_bytes);
443            le_bytes
444                .into_iter()
445                .flat_map(|byte| (0..8).map(move |shift| (byte >> shift) % 2 == 1))
446        };
447
448        // Take the first 255 bits.
449        bitstring
450            .take(pallas::Scalar::NUM_BITS as usize)
451            .collect::<Vec<_>>()
452    });
453
454    // Transpose.
455    let mut bitstring = bitstring.transpose_vec(pallas::Scalar::NUM_BITS as usize);
456    // Reverse to get the big-endian bit representation.
457    bitstring.reverse();
458    bitstring
459}

772            fn synthesize(
773                &self,
774                config: MulConfig,
775                mut layouter: impl Layouter<pallas::Base>,
776            ) -> Result<(), Error> {
777                let chip = EccChip::construct(config.ecc.clone(), self.circuit_version);
778                let base = NonIdentityPoint::new(
779                    chip.clone(),
780                    layouter.namespace(|| "base"),
781                    Value::known(pallas::Point::generator().to_affine()),
782                )?;
783                let scalar = chip.load_private(
784                    layouter.namespace(|| "scalar"),
785                    config.ecc.advices[0],
786                    Value::known(pallas::Base::from(7)),
787                )?;
788                let scalar =
789                    ScalarVar::from_base(chip, layouter.namespace(|| "scalar var"), &scalar)?;
790                base.mul(layouter.namespace(|| "[scalar] base"), scalar)?;
791                Ok(())
792            }

797        /// Synthesizes `MulCircuit` for the given version through a [`CopyRecorder`].
798        fn record(circuit_version: CircuitVersion) -> CopyRecorder {
799            let circuit = MulCircuit { circuit_version };
800            let mut cs = ConstraintSystem::<pallas::Base>::default();
801            let config = MulCircuit::configure(&mut cs);
802            let constants = vec![config.constants];
803            let mut recorder = CopyRecorder::default();
804            <MulCircuit as Circuit<pallas::Base>>::FloorPlanner::synthesize(
805                &mut recorder,
806                &circuit,
807                config,
808                constants,
809            )
810            .expect("synthesis should succeed");
811            recorder
812        }

814        #[test]
815        fn fixed_circuit_anchors_incomplete_base() {
816            let insecure: BTreeSet<Copy> = record(CircuitVersion::InsecureUnanchoredBase)
817                .copies
818                .into_iter()
819                .collect();
820            let fixed_recorder = record(CircuitVersion::AnchoredBase);
821            let fixed: BTreeSet<Copy> = fixed_recorder.copies.iter().cloned().collect();
822
823            // The fix only adds copy constraints; it changes no layout. So every copy the
824            // insecure circuit emits must still be present in the fixed circuit.
825            assert!(
826                insecure.is_subset(&fixed),
827                "the fixed circuit must retain every copy constraint of the insecure circuit"
828            );
829
830            // `configure` is deterministic, so this config's columns/selectors match the run.
831            let config = MulCircuit::configure(&mut ConstraintSystem::default());
832            let mul = &config.ecc.mul;
833            let x_p = Column::<Any>::from(mul.hi_config.double_and_add.x_p);
834            let y_p = Column::<Any>::from(mul.hi_config.y_p);
835
836            // The anchor must sit on the first incomplete-addition row — the row where the
837            // `(x_p, y_p)`-constancy gate `q_mul_2` first fires — so that constancy propagates
838            // the base to every subsequent incomplete row. (Anchoring one row earlier, on the
839            // `q_mul_1` doubling row, would not be reached by the constancy chain.)
840            let first_incomplete_row = fixed_recorder
841                .selectors
842                .iter()
843                .filter(|(s, _)| *s == mul.hi_config.q_mul_2)
844                .map(|&(_, row)| row)
845                .min()
846                .expect("q_mul_2 must be enabled");
847
848            // The `hi` and `lo` halves share the `x_p`/`y_p` columns (see `Config::configure`) and
849            // anchor at the same offset, so the four `copy_advice` calls collapse to exactly two
850            // new equality constraints. Each is recorded as `(dst_col, dst_row, src_col, src_row)`:
851            // the `copy_advice` writes the anchored cell at `(x_p|y_p, first_incomplete_row)` and
852            // equates it to the source base coordinate, which this circuit witnesses in the same
853            // column at row 0.
854            let extra: BTreeSet<Copy> = fixed.difference(&insecure).cloned().collect();
855            assert_eq!(
856                extra,
857                [
858                    (x_p, first_incomplete_row, x_p, 0),
859                    (y_p, first_incomplete_row, y_p, 0),
860                ]
861                .into_iter()
862                .collect()
863            );
864        }
